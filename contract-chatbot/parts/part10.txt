function norm(s){
  return (s||"").normalize("NFKC").toLowerCase()
    .replace(/[·ㆍ․]/g," ")
    .replace(/[^\p{L}\p{N}.%]+/gu," ")
    .replace(/\s+/g," ").trim();
}
function compact(s){ return norm(s).replace(/\s/g,""); }
function tokenize(s){
  return norm(s).split(/\s+/).filter(t => t.length>1 && !STOP.has(t));
}
function expandTerms(question){
  const n = compact(question);
  const out = new Set(tokenize(question));
  for(const [key,vals] of Object.entries(ALIASES)){
    if(n.includes(compact(key)) || vals.some(v => n.includes(compact(v)))){
      vals.forEach(v=>out.add(norm(v)));
    }
  }
  return [...out];
}
function occ(hay, needle){
  if(!needle || needle.length<2) return 0;
  let c=0, p=0;
  while((p=hay.indexOf(needle,p))!==-1){c++;p+=needle.length;}
  return c;
}
